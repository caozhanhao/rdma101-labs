cmake_minimum_required(VERSION 3.20)
project(rdma101 LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Build the Engine inside Linux. See docs/00-environment.md for a Linux VM and RXE setup.")
endif()
find_package(Threads REQUIRED)
find_path(VERBS_INCLUDE infiniband/verbs.h REQUIRED)
find_library(VERBS_LIBRARY ibverbs REQUIRED)
add_library(rdma101_c SHARED engine.c)
target_compile_features(rdma101_c PRIVATE c_std_17)
target_include_directories(rdma101_c PRIVATE .. ${VERBS_INCLUDE})
target_compile_options(rdma101_c PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(rdma101_c PRIVATE Threads::Threads ${VERBS_LIBRARY})
