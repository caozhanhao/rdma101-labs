cmake_minimum_required(VERSION 3.20)
project(rdma101 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Build the Engine inside Linux. See docs/00-environment.md for a Linux VM and RXE setup.")
endif()
find_package(Threads REQUIRED)
find_path(VERBS_INCLUDE infiniband/verbs.h REQUIRED)
find_library(VERBS_LIBRARY ibverbs REQUIRED)
add_library(rdma101_cpp SHARED engine.cpp ffi.cpp)
target_compile_features(rdma101_cpp PRIVATE cxx_std_20)
target_include_directories(rdma101_cpp PRIVATE .. ${VERBS_INCLUDE})
target_compile_options(rdma101_cpp PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(rdma101_cpp PRIVATE Threads::Threads ${VERBS_LIBRARY})
